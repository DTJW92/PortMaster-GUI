CUR_TTY=/dev/tty0

export directory="userdata/roms"

## HippOS runs ports, their games and PortMaster as its own user, so everything they write
## stays the user's. Only what needs root goes through its passwordless sudo: mounts, device
## and service changes, and file operations on system paths.
pm_hippos_sudo() {
  sudo --preserve-env=SDL_GAMECONTROLLERCONFIG_FILE,DEVICE,param_device,HOTKEY,ANALOGSTICKS "$@"
}

pm_hippos_system_paths() {
  local arg
  for arg in "$@"; do
    case "$arg" in
      /dev/shm/*) ;;
      /sys/*|/proc/*|/dev/*|/etc/*|/usr/*|/opt/*|/boot/*|/lib*/*|/var/*) return 0 ;;
    esac
  done
  return 1
}

pm_hippos_esudo() {
  local i=1
  if [ "$1" = "env" ]; then
    i=2
    while [ "$i" -le "$#" ]; do
      case "${!i}" in
        -u) i=$((i + 2)) ;;
        -*|*=*) i=$((i + 1)) ;;
        *) break ;;
      esac
    done
  fi

  case "${!i##*/}" in
    mount|umount|chmod|chown|systemctl|modprobe|sysctl|swapon|swapoff|mkswap)
      pm_hippos_sudo "$@"; return $? ;;
    mkdir|rm|tee|tar|cp|mv|touch|ln)
      if pm_hippos_system_paths "${@:i+1}"; then
        pm_hippos_sudo "$@"; return $?
      fi ;;
  esac

  "$@"
}
export ESUDO="pm_hippos_esudo"
export ESUDOKILL="-1"
export ESUDOKILL2="-1"

## HippOS's launcher passes the player's mappings in SDL_GAMECONTROLLERCONFIG.
if [ -n "$SDL_GAMECONTROLLERCONFIG" ]; then
  printf '%s\n' "$SDL_GAMECONTROLLERCONFIG" > "$XDG_RUNTIME_DIR/gamecontrollerdb.txt"
  export SDL_GAMECONTROLLERCONFIG_FILE="$XDG_RUNTIME_DIR/gamecontrollerdb.txt"
elif [ -f "$HOME/.config/gamecontrollerdb.txt" ]; then
  export SDL_GAMECONTROLLERCONFIG_FILE="$HOME/.config/gamecontrollerdb.txt"
else
  export SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.txt"
fi

SDLDBFILE="${SDL_GAMECONTROLLERCONFIG_FILE}"
SDLDBUSERFILE="${HOME}/.config/SDL-GameControllerDB/gamecontrollerdb.txt"
[ ! -f "${SDLDBUSERFILE}" ] && SDLDBUSERFILE="$SDL_GAMECONTROLLERCONFIG_FILE"

get_controls() {
  ANALOGSTICKS="${ANALOG_STICKS:-2}"
  LOWRES="N"

  sdl_controllerconfig="$(< "${SDL_GAMECONTROLLERCONFIG_FILE}")"
  ## With no controller the file is the whole db: as SDL_GAMECONTROLLERCONFIG it passes
  ## MAX_ARG_STRLEN and every exec fails (E2BIG). SDL still reads SDL_GAMECONTROLLERCONFIG_FILE.
  [ "${#sdl_controllerconfig}" -gt 100000 ] && sdl_controllerconfig=""
}

if [ -e "/lib64/ld-linux-x86-64.so.2" ] || [ -e "/lib/ld-linux-x86-64.so.2" ]; then
  if [ -f "$controlfolder/gptokeyb.x86_64" ]; then
    mv "$controlfolder/gptokeyb.x86_64" "$controlfolder/gptokeyb"
  fi

  if [ -f "$controlfolder/gptokeyb2.x86_64" ]; then
    mv "$controlfolder/gptokeyb2.x86_64" "$controlfolder/gptokeyb2"
  fi

  if [ -f "$controlfolder/xdelta3.x86_64" ]; then
    mv "$controlfolder/xdelta3.x86_64" "$controlfolder/xdelta3"
  fi
fi

source $controlfolder/device_info.txt
source $controlfolder/funcs.txt

$ESUDO chmod 666 /dev/tty0
$ESUDO chmod 666 /dev/tty1
$ESUDO chmod 666 /dev/uinput

export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH}.so $controlfolder/gptokeyb2 $ESUDOKILL2"
export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
